## routing rules to agents
kubectl exec -n oai deployment/xchain-orchestrator -- python3 -c "
import urllib.request, json

# Override tractor chain to hit the enhancer port
for name, host, port in [
    ('tractor',   'tractor-mono',     4300),
    ('fastinfer', 'xchain-fastinfer', 4400),
    ('null',      'null-sink',        1),
]:
    data = json.dumps({'name': name, 'host': host, 'port': port}).encode()
    req = urllib.request.Request(
        'http://localhost:5010/chains',
        data=data,
        headers={'Content-Type': 'application/json'},
        method='POST'
    )
    print(urllib.request.urlopen(req).read().decode())
"
